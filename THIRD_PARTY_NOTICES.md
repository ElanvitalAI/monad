## Regenerating this file

```bash
# dependency table
bun install                      # node_modules must be present
python3 - <<'EOF'
import json, os
d = json.load(open('package.json'))
for name, ver in sorted(d['dependencies'].items()):
    p = os.path.join('node_modules', *name.split('/'), 'package.json')
    try:    lic = json.load(open(p)).get('license') or 'UNDECLARED'
    except Exception: lic = 'UNDECLARED'
    print(f'{lic}\t{name}\t{ver}')
EOF

# in-source attribution count
rg -ci 'ported from|adapted from|derived from|copied from' \
   src/ scripts/ skills/ --glob '!*.test.ts' | awk -F: '{s+=$NF} END{print s+0}'
```
